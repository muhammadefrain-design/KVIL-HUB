-- Logic helpers
local function setR6Anim(walkId, runId)
	local char = LocalPlayer.Character
	if not char then return end
	local animate = char:FindFirstChild("Animate")
	if not animate then return end
	pcall(function()
		if animate:FindFirstChild("walk") and animate.walk:FindFirstChild("WalkAnim") then
			animate.walk.WalkAnim.AnimationId = walkId
		end
		if animate:FindFirstChild("run") and animate.run:FindFirstChild("RunAnim") then
			animate.run.RunAnim.AnimationId = runId
		end
	end)
end

local function PlayAnim(name)
	local data = R6_ANIMS[name]
	if data then setR6Anim(data.Walk, data.Run) end
end
local function StopAnim()
	setR6Anim(R6_DEFAULT_WALK, R6_DEFAULT_RUN)
end

local function GetHead()
	local root = LocalPlayer.Character and LocalPlayer.Character:FindFirstChild("HumanoidRootPart")
	if not root then return nil end
	local best, short = nil, 250
	for _, plr in ipairs(Players:GetPlayers()) do
		if not (plr == LocalPlayer) and plr.Character then
			local head = plr.Character:FindFirstChild("Head")
			local hum = plr.Character:FindFirstChildOfClass("Humanoid")
			if head and hum and hum.Health > 0 then
				local d = (head.Position - root.Position).Magnitude
				if d < short then short = d best = head end
			end
		end
	end
	return best
end

local function MasCercano(maxD)
	local root = LocalPlayer.Character and LocalPlayer.Character:FindFirstChild("HumanoidRootPart")
	if not root then return nil end
	local best, bestD = nil, maxD
	for _, plr in ipairs(Players:GetPlayers()) do
		if not (plr == LocalPlayer) and plr.Character then
			local r = plr.Character:FindFirstChild("HumanoidRootPart")
			local h = plr.Character:FindFirstChildOfClass("Humanoid")
			if r and h and h.Health > 0 then
				local d = (r.Position - root.Position).Magnitude
				if d < bestD then bestD = d best = plr end
			end
		end
	end
	return best
end

local function EquiparArma()
	local char = LocalPlayer.Character
	local hum = char and char:FindFirstChildOfClass("Humanoid")
	if not hum then return nil end
	local names = {"Gun", "gun", "Revolver", "Pistol", "Knife", "knife", "Cuchillo"}
	for _, n in ipairs(names) do
		local t = char:FindFirstChild(n)
		if t and t:IsA("Tool") then return t end
	end
	local bp = LocalPlayer:FindFirstChild("Backpack")
	if bp then
		for _, n in ipairs(names) do
			local t = bp:FindFirstChild(n)
			if t and t:IsA("Tool") then hum:EquipTool(t) return t end
		end
	end
	return nil
end

local hitboxParts = {}
local function ClearHitbox()
	for _, parts in pairs(hitboxParts) do
		for _, p in ipairs(parts) do pcall(function() p:Destroy() end) end
	end
	hitboxParts = {}
end

local function ApplyHitbox()
	ClearHitbox()
	if not HitboxOn then return end
	for _, plr in ipairs(Players:GetPlayers()) do
		if not (plr == LocalPlayer) and plr.Character then
			local list = {}
			for _, part in ipairs(plr.Character:GetDescendants()) do
				if part:IsA("BasePart") and (part.Name == "Head" or part.Name == "HumanoidRootPart" or part.Name == "Torso" or part.Name == "UpperTorso") then
					local box = Instance.new("Part")
					box.Name = "HB_KV"
					box.Size = Vector3.new(HitboxSize, HitboxSize, HitboxSize)
					box.Transparency = 0.88
					box.Color = Color3.fromRGB(255, 60, 60)
					box.Material = Enum.Material.Neon
					box.CanCollide = false
					box.Massless = true
					box.CFrame = part.CFrame
					box.Parent = part
					local w = Instance.new("WeldConstraint")
					w.Part0 = part
					w.Part1 = box
					w.Parent = box
					table.insert(list, box)
				end
			end
			hitboxParts[plr] = list
		end
	end
end
